; ============================================================
; YSD8800 ISA2.1（正式確定版）
; ============================================================

; ------------------------------------------------------------
; 拡張プレフィックス
; ------------------------------------------------------------
0x1F : EXT

; ------------------------------------------------------------
; EXT サブオペコード
; ------------------------------------------------------------

; === Stack Operations ===
0x00 : PUSH A
0x01 : PUSH B
0x02 : PUSH X

0x03 : POP A
0x04 : POP B
0x05 : POP X


; === 8bit Load ===
0x10 : LDB A,[addr]   (imm16続く)
0x11 : LDB A,[X]

0x12 : LDB B,[addr]
0x13 : LDB B,[X]


; === 8bit Store ===
0x14 : STB A,[addr]   (imm16続く)
0x15 : STB A,[X]

0x16 : STB B,[addr]
0x17 : STB B,[X]


; ------------------------------------------------------------
; 動作仕様
; ------------------------------------------------------------

; PUSH
SP = SP - 2
mem[SP] = reg

; POP
reg = mem[SP]
SP = SP + 2

; LDB
reg = zero_extend(mem8)

; STB
mem8 = reg & 0xFF

; FLAGSは変更しない
